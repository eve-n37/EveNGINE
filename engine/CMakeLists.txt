cmake_minimum_required(VERSION 3.20)
project(EveNGINE VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(LOTA_DESKTOP "Build the SDL3 desktop backend" ON)
option(LOTA_TESTS "Build regression tests" OFF)
option(LOTA_GAME "Build the Legend of the Ancients RPG" OFF)
option(LOTA_SAMPLES "Build the original demo campaigns" OFF)
option(LOTA_STARTER "Build the standalone SDK starter" OFF)
option(LOTA_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(LOTA_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(lota_core src/core.cpp)
if(LOTA_SANITIZE AND NOT MSVC)
  target_compile_options(lota_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(lota_core INTERFACE -fsanitize=address,undefined)
endif()
add_library(LotA::Core ALIAS lota_core)
add_library(EveNGINE::Core ALIAS lota_core)
set_target_properties(lota_core PROPERTIES EXPORT_NAME Core)
target_compile_features(lota_core PUBLIC cxx_std_17)
target_include_directories(lota_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(MSVC)
  target_compile_options(lota_core PRIVATE /W4)
else()
  target_compile_options(lota_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(lota_studio src/studio.cpp src/narrative.cpp src/appearance.cpp src/release.cpp src/adventure.cpp)
add_library(LotA::Studio ALIAS lota_studio)
add_library(EveNGINE::Studio ALIAS lota_studio)
set_target_properties(lota_studio PROPERTIES EXPORT_NAME Studio)
target_link_libraries(lota_studio PUBLIC lota_core)

add_library(lota_platform)
add_library(LotA::Platform ALIAS lota_platform)
add_library(EveNGINE::Platform ALIAS lota_platform)
set_target_properties(lota_platform PROPERTIES EXPORT_NAME Platform)
target_link_libraries(lota_platform PUBLIC lota_core)
if(LOTA_DESKTOP)
  set(LOTA_SDL_FRAMEWORK "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL3.framework" CACHE PATH "Optional macOS SDL3 framework")
  if(NOT (APPLE AND EXISTS "${LOTA_SDL_FRAMEWORK}/SDL3"))
    find_package(SDL3 3.2 CONFIG QUIET)
  endif()
  if(NOT SDL3_FOUND AND APPLE AND EXISTS "${LOTA_SDL_FRAMEWORK}/SDL3")
    add_library(SDL3::SDL3 INTERFACE IMPORTED)
    set_target_properties(SDL3::SDL3 PROPERTIES INTERFACE_LINK_OPTIONS "-F${LOTA_SDL_FRAMEWORK}/..;SHELL:-framework SDL3")
    target_include_directories(lota_platform PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/include")
    target_link_options(lota_platform INTERFACE "$<BUILD_INTERFACE:-Wl,-rpath,${LOTA_SDL_FRAMEWORK}/..>" "$<BUILD_INTERFACE:-Wl,-rpath,@executable_path/../Frameworks>")
    install(DIRECTORY "${LOTA_SDL_FRAMEWORK}" DESTINATION Frameworks USE_SOURCE_PERMISSIONS)
  elseif(NOT SDL3_FOUND)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL3-3.2.20/CMakeLists.txt")
      message(FATAL_ERROR "SDL3 not found. Install SDL3, supply a macOS framework, or use -DLOTA_DESKTOP=OFF.")
    endif()
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(vendor/SDL3-3.2.20 EXCLUDE_FROM_ALL)
  endif()
  target_sources(lota_platform PRIVATE src/platform/sdl.cpp)
  target_link_libraries(lota_platform PUBLIC SDL3::SDL3)
else()
  target_sources(lota_platform PRIVATE src/platform/headless.cpp)
endif()
install(TARGETS lota_core lota_platform lota_studio EXPORT LotAEngineTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/lota/core.hpp include/lota/platform.hpp include/lota/json.hpp include/lota/studio.hpp include/lota/audio.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/lota)
install(EXPORT LotAEngineTargets NAMESPACE LotA:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LotAEngine)
configure_package_config_file(cmake/LotAEngineConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/LotAEngineConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LotAEngine)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/LotAEngineConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/LotAEngineConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/LotAEngineConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LotAEngine)
if(LOTA_TESTS)
  enable_testing()
endif()
if(LOTA_SAMPLES)
  add_library(lota_sample src/sample_game.cpp)
  target_link_libraries(lota_sample PUBLIC lota_core)
  add_executable(lota-engine src/main.cpp)
  target_link_libraries(lota-engine PRIVATE lota_sample lota_platform)
  add_custom_target(lota_assets ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/samples" "$<TARGET_FILE_DIR:lota-engine>/samples")
  add_dependencies(lota-engine lota_assets)
  if(LOTA_TESTS)
    add_executable(lota-tests tests/tests.cpp)
    target_link_libraries(lota-tests PRIVATE lota_sample)
    add_test(NAME engine-regression COMMAND lota-tests "${CMAKE_CURRENT_SOURCE_DIR}/samples" "${CMAKE_CURRENT_BINARY_DIR}")
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(UNIX AND NOT LOTA_SANITIZE AND Python3_Interpreter_FOUND)
      add_test(NAME engine-small-stack COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/small_stack.py" "$<TARGET_FILE:lota-engine>" "${CMAKE_CURRENT_SOURCE_DIR}/samples")
    endif()
  endif()
endif()
if(LOTA_GAME)
  add_library(lota_game src/game/game.cpp)
  target_link_libraries(lota_game PUBLIC lota_core)
  add_executable(lota-game src/game/main.cpp)
  target_link_libraries(lota-game PRIVATE lota_game lota_platform)
  add_custom_target(lota_game_assets ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/games" "$<TARGET_FILE_DIR:lota-game>/games")
  add_dependencies(lota-game lota_game_assets)
  if(LOTA_TESTS)
    add_executable(lota-game-tests tests/game_tests.cpp)
    target_link_libraries(lota-game-tests PRIVATE lota_game)
    add_test(NAME lota-gameplay COMMAND lota-game-tests "${CMAKE_CURRENT_SOURCE_DIR}/games/lota" "${CMAKE_CURRENT_BINARY_DIR}")
    add_executable(lota-render-tests tests/render_tests.cpp)
    target_link_libraries(lota-render-tests PRIVATE lota_game)
    add_test(NAME godot-geometry-parity COMMAND lota-render-tests "${CMAKE_CURRENT_SOURCE_DIR}/games/lota" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference/geometry.json" "${CMAKE_CURRENT_BINARY_DIR}/render-checks")
    add_executable(lota-ui-tests tests/ui_tests.cpp)
    target_link_libraries(lota-ui-tests PRIVATE lota_game)
    add_test(NAME ui-animation-settings COMMAND lota-ui-tests "${CMAKE_CURRENT_SOURCE_DIR}/games/lota" "${CMAKE_CURRENT_BINARY_DIR}/ui-checks")
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import PIL" RESULT_VARIABLE LOTA_PIL_RESULT OUTPUT_QUIET ERROR_QUIET)
      if(LOTA_PIL_RESULT EQUAL 0)
        add_test(NAME godot-visual-parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/visual_tests.py" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference" "${CMAKE_CURRENT_BINARY_DIR}/render-checks" "${CMAKE_CURRENT_SOURCE_DIR}/games/lota/world.json")
        set_tests_properties(godot-visual-parity PROPERTIES DEPENDS godot-geometry-parity)
      endif()
    endif()
  endif()
endif()
if(LOTA_STARTER)
  add_executable(lota-starter examples/starter/main.cpp)
  target_link_libraries(lota-starter PRIVATE lota_core lota_platform)
  add_custom_command(TARGET lota-starter POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/examples/starter/content" "$<TARGET_FILE_DIR:lota-starter>/starter/content")
  if(LOTA_TESTS)
    add_test(NAME sdk-starter-smoke COMMAND lota-starter --project "${CMAKE_CURRENT_SOURCE_DIR}/examples/starter/content/game.project" --smoke --ticks 30 --save "${CMAKE_CURRENT_BINARY_DIR}/starter-save.json" --capture "${CMAKE_CURRENT_BINARY_DIR}/starter.ppm")
  endif()
endif()

install(DIRECTORY include/evengine DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/EveNGINEConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/EveNGINEConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EveNGINE)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/EveNGINEConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EveNGINEConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/EveNGINEConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/EveNGINE)
